# CPU ONNX runtime. InsightFace does not share torch's CUDA context, and
# onnxruntime-gpu needs its own CUDA/cuDNN matching your torch build - a

# then set face.use_gpu: true. The code falls back to CPU with a warning if the
# CUDA execution provider is not actually available.
